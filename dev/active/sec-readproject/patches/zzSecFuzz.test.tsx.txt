function withDoc(mut: (doc: J) => void) {
  const rec: J = clone(seedDocRecord());
  mut(rec.doc);
  rec.doc.hash = hashDoc(rec.doc as PageDoc);
  rec.snapshots = rec.snapshots.map((s: J) => ({ ...s, doc: rec.doc, hash: rec.doc.hash }));
  return seedFile({ doc: rec });
}
const firstText = (doc: J) => {
  for (const s of doc.sections) for (const [k, v] of Object.entries(s.slots)) if (typeof v === "string") return [s, k] as const;
  throw new Error("no text slot");
};
const firstImage = (doc: J) => {
  for (const s of doc.sections) for (const [k, v] of Object.entries(s.slots)) if (v && typeof v === "object") return [s, k] as const;

};
const withSeries = (mut: (base: J, v: J) => void) => {
  const series: J = clone(seedSeries());
  series.forEach((v: J) => mut(v.base, v));
  return seedFile({ series });
};

const BAD = [/<script/i, /\son[a-z]+=/i, /javascript:/i, /url\(\s*['"]?https?:/i, /evil\.example/i];
const leaks = (html: string) => {
  const d = new DOMParser().parseFromString(html, "text/html");
  const hits: string[] = [];
  const scripts = [...d.querySelectorAll("script")].filter((s) => !s.closest("head"));
  if (scripts.length) hits.push(`script x${scripts.length}`);
  for (const el of d.querySelectorAll("*")) for (const a of el.attributes) {
    if (/^on/i.test(a.name)) hits.push(`attr ${a.name}`);
    if (/^(href|src|action|formaction|xlink:href)$/i.test(a.name) && /^\s*(javascript:|https?:)/i.test(a.value)) hits.push(`url ${a.name}=${a.value.slice(0, 30)}`);
    if (a.name === "style" && /url\(\s*['"]?https?:/i.test(a.value)) hits.push("style url");
  }
  if (d.querySelector("meta[http-equiv], base, link, iframe, object, embed, template")) hits.push("bad element");
  return hits;
};

async function probe(name: string, make: () => unknown) {
  const value = make();
  const file = typeof value === "string" && value.startsWith("RAW:") ? fileOf(value.slice(4)) : jsonFile(value);
  const t0 = performance.now();
  const checked = await checkFile(file, fakeDeps().deps);
  const ms = Math.round(performance.now() - t0);
  if (!checked.ok) return `${name} | REJECT ${checked.code} | ${ms}ms`;
  const out: string[] = [`${name} | PASS import | ${ms}ms`];
  const polluted = ({} as J).polluted !== undefined || ({} as J).hasDoc !== undefined;
  out.push(`protoPolluted=${polluted}`);
  const plan = rekeyImport(checked.file, undefined, "2026-10-18T00:00:00.000Z");
  if (!plan.ok) return `${out.join(" | ")} | rekey REJECT`;
  const state = mergeImport(undefined, plan.plan);
  const stored = structuredClone(state.projects.get(plan.plan.projectId)) as J;
  out.push(`storedProjectKeys=${Object.keys(stored).join(",")}`);
  if (Object.getPrototypeOf(stored) !== Object.prototype) out.push("storedProjectProto=CHANGED");
  const doc = plan.plan.doc?.doc as PageDoc | undefined;
  const kit = docKitTokens({ profileId: plan.plan.profileId, versions: plan.plan.series } as J, doc?.profileVersion ?? plan.plan.series.length);
  const msg = readParentMessage({ type: "render", doc: doc ?? {}, ...(kit && { kitTokens: kit }) });
  out.push(`kitTokens=${kit ? "built" : "none"} renderMsg=${msg ? "accepted" : "DROPPED"}`);
  if (kit) {
    try {
      out.push(`siteFaces=${JSON.stringify(siteFaces(kit.type).map((f) => String(f.family)))} fontStack=${JSON.stringify(fontStack(kit.type.family)).slice(0, 80)}`);
      const vars = kitVars(kit);
      out.push(`kitVarLeaks=${JSON.stringify(leaks(Object.values(vars).join(";")))}`);
    } catch (e) {
      out.push(`kit THROW ${(e as Error).message.slice(0, 60)}`);
    }
  }
  if (doc && msg) {
    try {
      const { container } = render(<PageDocument doc={doc} kitTokens={msg.type === "render" ? msg.kitTokens : undefined} />);
      const markup = container.innerHTML;
      out.push(`renderLeaks=${JSON.stringify(leaks(markup))}`);
      const html = buildStaticHtml({ markup: (container.firstElementChild as Element).outerHTML, css: "", title: doc.meta.title, description: doc.meta.description });
      out.push(`staticLeaks=${JSON.stringify(leaks(html.replace(/<script>[^]*?<\/script>/, "")))} titleEsc=${html.includes("&lt;/title&gt;") || !doc.meta.title.includes("<")}`);
    } catch (e) {
      out.push(`render/static THROW ${(e as Error).message.slice(0, 80)}`);
    } finally {
      cleanup();
    }
  }
  return out.join(" | ");
}

describe("SEC fuzz", () => {
  it("runs all cases", async () => {
    const lines: string[] = [];
    for (const [name, make] of CASES) lines.push(await probe(name, make).catch((e) => `${name} | PROBE THROW ${(e as Error).message.slice(0, 100)}`));
    (await import("node:fs")).writeFileSync(process.env.SECFUZZ_OUT ?? "/tmp/secfuzz.txt", lines.join("\n") + "\n");
    expect(lines.length).toBe(CASES.length);
  }, 120_000);
});
